whitespace:
    [\n\t\r\v\f]

comments:
    \\ is an in-line comment

identifiers:
    [a-zA-Z_][a-zA-Z_0-9]*   <--- 0 or more

keywords:
    let, break, continue, return, loop, if, else

symbol tokens:
    ; ~ ^ * % ( ) : { } [ ] + - ,
    | ||
    & &&
    = ==
    ! !=
    < << <=
    > >> >=
    /   <--- Note that // is a comment but / is a division

integer literals:
    decimal: [0-9]+  <--- 1 or more
    (
        -> Note that all integer literals are parsed as positive numbers
        -> -343 is parsed as unary minus and 343
        -> if the value would exceed a 132's max value: compilation must fail
    )
